import os, json, threading, asyncio
from flask import Flask
from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.ext import Application, CommandHandler, CallbackQueryHandler, ContextTypes, ConversationHandler, MessageHandler, filters

TOKEN = os.getenv("BOT_TOKEN")
TITLE, Q_TEXT, Q_OPT, Q_ANS = range(4)
QUIZ_FILE = "quizzes.json"

def load_q():
    if os.path.exists(QUIZ_FILE):
        with open(QUIZ_FILE, 'r', encoding='utf-8') as f: return json.load(f)
    return {}

def save_q(data):
    with open(QUIZ_FILE, 'w', encoding='utf-8') as f: json.dump(data, f, ensure_ascii=False, indent=2)

quizzes = load_q()
user_scores = {}

flask_app = Flask(__name__)
@flask_app.route('/')
def home(): return "Quiz Maker Bot LIVE!"

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        "🔥 ADVANCE QUIZ MAKER BOT\n\n"
        "/newquiz - Naya Quiz Banao\n"
        "/myquiz - Aapke Banaye Quiz\n"
        "/quiz - Quiz Khelo (group me)\n"
        "/leaderboard - Score\n\n"
        "Pehle /newquiz se quiz banao!"
    )

# --- QUIZ BANANA ---
async def newquiz(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("Quiz ka Naam / Title bhejo:\nEx: Bihar GK 2026")
    return TITLE

async def get_title(update: Update, context: ContextTypes.DEFAULT_TYPE):
    context.user_data['new_quiz'] = {"title": update.message.text, "questions": [], "owner": update.effective_user.id}
    context.user_data['qid'] = f"{update.effective_user.id}_{len(quizzes)}"
    await update.message.reply_text("Ab Pehla Sawaal bhejo:")
    return Q_TEXT

async def get_q_text(update: Update, context: ContextTypes.DEFAULT_TYPE):
    context.user_data['curr_q'] = {"q": update.message.text}
    await update.message.reply_text("Ab 4 Options bhejo alag alag line me:\nEx:\nPatna\nGaya\nBhagalpur\nPurnia")
    return Q_OPT

async def get_q_opt(update: Update, context: ContextTypes.DEFAULT_TYPE):
    opts = update.message.text.split("\n")
    if len(opts) < 2:
        await update.message.reply_text("Kam se kam 2 option bhejo line break karke")
        return Q_OPT
    context.user_data['curr_q']['o'] = opts[:4]
    await update.message.reply_text(f"Ab Sahi Jawab ka Number bhejo (1-{len(opts)}):")
    return Q_ANS

async def get_q_ans(update: Update, context: ContextTypes.DEFAULT_TYPE):
    try:
        ans = int(update.message.text) - 1
        curr = context.user_data['curr_q']
        if ans < 0 or ans >= len(curr['o']): raise ValueError
        curr['a'] = ans

        new_quiz = context.user_data['new_quiz']
        new_quiz['questions'].append(curr)

        # Save
        qid = context.user_data['qid']
        quizzes[qid] = new_quiz
        save_q(quizzes)

        keyboard = [[InlineKeyboardButton("➕ Aur Sawaal Add Karo", callback_data="more")],
                    [InlineKeyboardButton("✅ Quiz Ready Karo", callback_data="done")]]
        await update.message.reply_text(f"Sawaal {len(new_quiz['questions'])} Add ho gaya! Aur add karna hai?", reply_markup=InlineKeyboardMarkup(keyboard))
        return ConversationHandler.END
    except:
        await update.message.reply_text("Sahi number bhejo 1-4")
        return Q_ANS

async def more_q_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    q = update.callback_query
    await q.answer()
    if q.data == "more":
        await q.edit_message_text("Agla Sawaal bhejo:")
        context.user_data['curr_q'] = {}
        # next question flow manually
        context.user_data['state'] = Q_TEXT
        # we need to set conversation again
        return Q_TEXT
    else:
        await q.edit_message_text(f"✅ Quiz Ban Gaya! ID: {context.user_data['qid']}\nGroup me /quiz {context.user_data['qid']} likh ke chalao")
        return ConversationHandler.END

# For more question, we use simple message handler
async def add_more_text(update: Update, context: ContextTypes.DEFAULT_TYPE):
    state = context.user_data.get('state')
    if state == Q_TEXT:
        context.user_data['curr_q'] = {"q": update.message.text}
        await update.message.reply_text("4 Options bhejo line me:")
        context.user_data['state'] = Q_OPT
    elif state == Q_OPT:
        opts = update.message.text.split("\n")
        context.user_data['curr_q']['o'] = opts[:4]
        await update.message.reply_text(f"Sahi ka number bhejo (1-{len(opts)}):")
        context.user_data['state'] = Q_ANS
    elif state == Q_ANS:
        try:
            ans = int(update.message.text)-1
            context.user_data['curr_q']['a'] = ans
            qid = context.user_data['qid']
            quizzes[qid]['questions'].append(context.user_data['curr_q'])
            save_q(quizzes)
            keyboard = [[InlineKeyboardButton("➕ Aur Sawaal", callback_data="more2")],[InlineKeyboardButton("✅ Bas Ho Gaya", callback_data="done2")]]
            await update.message.reply_text(f"Add ho gaya! Total {len(quizzes[qid]['questions'])}", reply_markup=InlineKeyboardMarkup(keyboard))
            context.user_data['state'] = None
        except:
            await update.message.reply_text("Number galat hai")

async def more2_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    q = update.callback_query
    await q.answer()
    if "more" in q.data:
        await q.edit_message_text("Agla Sawaal bhejo:")
        context.user_data['state'] = Q_TEXT
        context.user_data['curr_q'] = {}
    else:
        await q.edit_message_text(f"✅ Quiz Ready! Group me jaake /quiz {context.user_data['qid']} bhejo")

async def myquiz(update: Update, context: ContextTypes.DEFAULT_TYPE):
    my = {k:v for k,v in quizzes.items() if str(v.get('owner')) == str(update.effective_user.id)}
    if not my:
        await update.message.reply_text("Aapne koi quiz nahi banaya. /newquiz bhejo")
        return
    txt = "Aapke Quiz:\n\n"
    for k,v in my.items():
        txt += f"ID: {k}\nTitle: {v['title']} ({len(v['questions'])} Q)\n/quiz {k} se chalao\n\n"
    await update.message.reply_text(txt)

# --- QUIZ KHELNA ---
async def send_q(chat_id, context, quiz_id, q_idx):
    q = quizzes[quiz_id]['questions'][q_idx]
    btns = [[InlineKeyboardButton(f"{opt}", callback_data=f"play_{quiz_id}_{q_idx}_{i}")] for i, opt in enumerate(q['o'])]
    await context.bot.send_message(chat_id, f"Q{q_idx+1}: {q['q']}", reply_markup=InlineKeyboardMarkup(btns))

async def quiz_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not quizzes:
        await update.message.reply_text("Koi quiz nahi hai. Pehle /newquiz se banao")
        return
    # agar /quiz ID ke saath hai
    qid = context.args[0] if context.args else list(quizzes.keys())[-1]
    if qid not in quizzes:
        await update.message.reply_text("ID galat hai. /myquiz dekho")
        return
    context.chat_data['active_quiz'] = qid
    context.chat_data['q_idx'] = 0
    await send_q(update.effective_chat.id, context, qid, 0)

async def play_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    q = update.callback_query
    await q.answer()
    _, quiz_id, q_idx, opt_idx = q.data.split("_")
    q_idx, opt_idx = int(q_idx), int(opt_idx)

    correct = quizzes[quiz_id]['questions'][q_idx]['a']
    user = q.from_user
    if user.id not in user_scores: user_scores[user.id] = {"name": user.first_name, "pts": 0}

    if opt_idx == correct:
        user_scores[user.id]['pts'] += 1
        res = "✅ Sahi!"
    else:
        res = f"❌ Galat! Sahi: {quizzes[quiz_id]['questions'][q_idx]['o'][correct]}"

    await q.edit_message_text(f"Q{q_idx+1}: {quizzes[quiz_id]['questions'][q_idx]['q']}\n\n{res}")
    await asyncio.sleep(1.5)
    nxt = q_idx + 1
    if nxt < len(quizzes[quiz_id]['questions']):
        await send_q(q.message.chat_id, context, quiz_id, nxt)
    else:
        await context.bot.send_message(q.message.chat_id, "🎉 Quiz Khatam! /leaderboard")

async def leaderboard(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not user_scores:
        await update.message.reply_text("Koi score nahi")
        return
    top = sorted(user_scores.values(), key=lambda x: x['pts'], reverse=True)[:10]
    txt = "🏆 Leaderboard\n\n" + "\n".join([f"{i+1}. {s['name']} - {s['pts']}" for i, s in enumerate(top)])
    await update.message.reply_text(txt)
